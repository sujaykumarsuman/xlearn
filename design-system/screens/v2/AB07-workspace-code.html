Solution

O(n²) time · O(1) extra space

Sorting lets you fix the smallest value of a triple and find the other two in one linear pass. For each first value nums[i] — skipping repeats — start one index just after it and one at the end. If the sum is below zero, move the left index right; above zero, move the right index left; at zero, record the triple and step past equal values.

Each first value costs O(n), so the search is O(n²) overall. Sorting in place needs no extra space beyond the answer itself.

GoC++Python
func threeSum(nums []int) [][]int {
    sort.Ints(nums)
    res := [][]int{}
    for i := 0; i+2 < len(nums); i++ {
        if i > 0 && nums[i] == nums[i-1] {
            continue // same first value
        }
        lo, hi := i+1, len(nums)-1
        for lo < hi {
            s := nums[i] + nums[lo] + nums[hi]
            if s < 0 {
                lo++
            } else if s > 0 {
                hi--
            } else {
                v := nums[lo]
                res = append(res, []int{nums[i],
                    v, nums[hi]})
                for lo < hi && nums[lo] == v {
                    lo++
                }
                hi--
            }
        }
    }
    return res
}
